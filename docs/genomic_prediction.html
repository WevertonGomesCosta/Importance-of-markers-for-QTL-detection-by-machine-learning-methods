Machine Learning for QTL Detection

This script performs genomic prediction using various Machine Learning algorithms. The primary goal is to extract Marker Importance Scores to evaluate their ability to detect QTLs compared to GWAS methods.

Models: Random Forest (RF), Bagging, Boosting (GBM), Decision Tree (DT), MARS.

1. Setup and Packages

We load the necessary libraries for data manipulation, parallel processing, and specific Machine Learning algorithms.

# Data Manipulation & Parallelism
library(tidyverse)  # Data handling
library(gtools)     # Natural sorting of files
library(doParallel) # Parallel backend
library(foreach)    # Parallel loops

# Machine Learning Models - Install if missing
if(!require(randomForest)) install.packages("randomForest") # RF & Bagging
if(!require(gbm)) install.packages("gbm")                   # Boosting (Gradient Boosting Machine)
if(!require(earth)) install.packages("earth")               # MARS (Multivariate Adaptive Regression Splines)
if(!require(rpart)) install.packages("rpart")               # Decision Trees

# Load ML Libraries
library(randomForest)
library(gbm)
library(earth)
library(rpart)

2. Data Preparation

We identify and list the partitioned files located in the PHEN_T (Phenotype) and GEN_T (Genotype) directories. We use natural sorting to ensure correct file ordering and perform integrity checks to guarantee that phenotype and genotype files are correctly paired.

# 1. Define File Paths
pathFenT <- file.path("data/PHEN_T")
pathGenT <- file.path("data/GEN_T")

# 2. File Listing (Natural Numeric Sort)
# 'mixedsort' ensures that "file10" comes after "file2", rather than after "file1"
filesFenT <- gtools::mixedsort(list.files(path = pathFenT, pattern = "\\.txt$", full.names = TRUE))
filesGenT <- gtools::mixedsort(list.files(path = pathGenT, pattern = "\\.txt$", full.names = TRUE))

# Integrity Check
if(length(filesFenT) == 0 || length(filesGenT) == 0) {
  stop("CRITICAL ERROR: No phenotype or genotype files found in data/ folders.")
}

if(length(filesFenT) != length(filesGenT)) {
  warning("WARNING: Mismatch in number of phenotype and genotype files. Check pairing.")
}

4. Machine Learning Execution

4.1. Multivariate Adaptive Regression Splines (MARS)

MARS is a non-parametric regression technique that models non-linear relationships by fitting piecewise linear segments (hinge functions). It automatically performs feature selection by “pruning” terms that do not contribute to model stability.

# 1. Setup
grau_mars <- 1:3 
grid_mars <- expand.grid(Rep = 1:n_reps, Variable = target_vars, Degree = grau_mars)
out_mars <- "output/mars"
if (!dir.exists(out_mars)) dir.create(out_mars, recursive = TRUE)

cat("Iniciando MARS. Total de tarefas:", nrow(grid_mars) * nfolds, "\n")

set.seed(123) 

# 2. Execução
mars_list <- foreach(task_i = 1:nrow(grid_mars), 
                     .packages = c('earth', 'caret', 'tidyverse'),
                     .export = c("filesFenT", "filesGenT", "filesFenV", "filesGenV", "nfolds")) %dopar% {
  
  curr_rep <- grid_mars$Rep[task_i]
  curr_var <- grid_mars$Variable[task_i]
  curr_deg <- grid_mars$Degree[task_i]
  
  # Definir nome composto do modelo
  model_name_compound <- paste("MARS", curr_deg)
  
  # Acumuladores locais (opcional, pois salvamos CSV)
  fold_metrics_accum <- data.frame()
  fold_importance_accum <- data.frame()
  
  for (fold_k in 1:nfolds) {
    
    # A. Lógica de Reciclagem de Arquivos
    raw_idx <- (curr_rep - 1) * nfolds + fold_k
    file_idx <- ((raw_idx - 1) %% length(filesFenT)) + 1
    
    # Checkpoint: Pula se já existir o arquivo de importância
    outfile_imp <- sprintf("%s/IMP_Rep%d_Var%d_Deg%d_Fold%d.csv", out_mars, curr_rep, curr_var, curr_deg, fold_k)
    if(file.exists(outfile_imp)) next
    
    tryCatch({
      # B. Leitura dos Dados (Sem renomear colunas)
      # O R usará o padrão: V1, V2, V3...
      dadosTy <- read.table(filesFenT[file_idx], header = FALSE)
      dadosTx <- read.table(filesGenT[file_idx], header = FALSE)
      dadosVy <- read.table(filesFenV[file_idx], header = FALSE)
      dadosVx <- read.table(filesGenV[file_idx], header = FALSE)
      
      start_time <- Sys.time()
      
      # C. Modelo
      mars_model <- earth(x = dadosTx, y = dadosTy[, curr_var], degree = curr_deg)
      
      exec_time <- as.numeric(difftime(Sys.time(), start_time, units = "secs"))
      
      # --- D. CÁLCULO DE MÉTRICAS ---
      ypred <- predict(mars_model, dadosVx)[,1] # [,1] garante vetor
      ypred_train <- mars_model$fitted.values[,1]
      
      # Validação
      if (sd(ypred) == 0) { 
        r2_val <- 0 
      } else { 
        rv <- cor(dadosVy[, curr_var], ypred)
        r2_val <- as.numeric(rv^2)
      }
      rmse_val <- as.numeric(sqrt(mean((dadosVy[, curr_var] - ypred)^2)))
      
      # Treinamento
      rt <- cor(dadosTy[, curr_var], ypred_train)
      r2_train <- as.numeric(rt^2)
      rmse_train <- as.numeric(sqrt(mean((dadosTy[, curr_var] - ypred_train)^2)))
      
      # Salvar Métricas
      curr_metrics <- data.frame(
        Model = model_name_compound, 
        Replicate = curr_rep, 
        Variable = curr_var, 
        Degree = curr_deg, 
        Fold = fold_k, 
        R2_Train = r2_train, 
        R2_Val = r2_val, 
        RMSE_Train = rmse_train, 
        RMSE_Val = rmse_val, 
        Time_Sec = exec_time
      )
      
      write.csv(curr_metrics, sprintf("%s/METRICS_Rep%d_Var%d_Deg%d_Fold%d.csv", out_mars, curr_rep, curr_var, curr_deg, fold_k), row.names = FALSE)
      
      # --- E. Importância (Código Solicitado) ---
      
      imp <- tryCatch({ data.frame(varImp(mars_model, value = "rss")) }, error=function(e) NULL)
      
      if (!is.null(imp)) {
         # --- PADRONIZAÇÃO DE COLUNAS (IMPORTÂNCIA) ---
         imp$Marker <- rownames(imp)
         # Se quiser remover o "V" padrão do R (V1 -> 1), descomente abaixo:
         # imp$Marker <- gsub("V", "", imp$Marker) 
         
         colnames(imp)[1] <- "Overall"
         
         imp_final <- data.frame(
           Model = model_name_compound,
           Replicate = curr_rep,
           Variable = curr_var,
           Degree = curr_deg,
           Fold = fold_k,
           Marker = imp$Marker,
           Overall = imp$Overall # Usando "rss" como medida de importância 
         )
         
         write.csv(imp_final, outfile_imp, row.names = FALSE)
      } else {
         # Salva placeholder vazio se nenhuma variável for selecionada
         dummy <- data.frame(Model=model_name_compound, Overall=0)
         write.csv(dummy, outfile_imp, row.names=FALSE)
      }
      
    }, error = function(e) {
       writeLines(paste("Erro no MARS:", e$message), file.path(out_mars, "error_log.txt"))
    })
    
    # Limpeza de memória
    rm(mars_model, dadosTx, dadosVx, dadosTy, dadosVy); gc()
  }
  return(NULL)
}

# 3. Consolidação Final (Executada após o loop paralelo)

metric_files <- list.files(out_mars, pattern = "METRICS_.*\\.csv", full.names = TRUE)

4.2. Decision Tree (DT)

Decision Trees partition the genetic data into recursive binary splits. The importance of a marker is calculated based on the improvement in the splitting criterion (e.g., ANOVA for regression) summed over all nodes where that marker was used.

grid_dt <- expand.grid(Rep = 1:n_reps, Variable = target_vars, stringsAsFactors = FALSE)

  curr_var <- grid_dt$Variable[task_i]

    # A. Lógica de Reciclagem de Arquivos
    raw_idx <- (curr_rep - 1) * nfolds + fold_k
    file_idx <- ((raw_idx - 1) %% length(filesFenT)) + 1

    if(file.exists(outfile_imp)) next
    
    tryCatch({
      # B. Leitura dos Dados

      # Salvar Métricas
      curr_metrics <- data.frame(
        Model = "DT", 
        Replicate = curr_rep, 
        Variable = curr_var, 
        Fold = fold_k, 
        R2_Train = r2_train, 
        R2_Val = r2_val, 
        RMSE_Train = rmse_train, 
        RMSE_Val = rmse_val,

      )
      
      write.csv(curr_metrics, sprintf("%s/METRICS_Rep%d_Var%d_Fold%d.csv", out_dt, curr_rep, curr_var, fold_k), row.names = FALSE)

      imp <- tryCatch({ data.frame(varImp(fit_tree, value = "rss")) }, error=function(e) NULL)

4.3. Bagging (BA)

Bagging is essentially a Random Forest where mtry (number of variables randomly sampled as candidates at each split) is equal to the total number of predictors (ncol(X)). This forces the trees to consider all markers at every split, reducing bias but potentially increasing correlation between trees compared to standard RF.

# 1. Setup
grid_ba <- expand.grid(Rep = 1:n_reps, Variable = target_vars, stringsAsFactors = FALSE)
out_ba <- "output/bagging"
if (!dir.exists(out_ba)) dir.create(out_ba, recursive = TRUE)

  curr_var <- grid_ba$Variable[task_i]
  
  # Acumuladores locais
  fold_metrics_accum <- data.frame()
  fold_importance_accum <- data.frame()
  
  for (fold_k in 1:nfolds) {
    
    # A. Lógica de Reciclagem de Arquivos
    raw_idx <- (curr_rep - 1) * nfolds + fold_k
    file_idx <- ((raw_idx - 1) %% length(filesFenT)) + 1

4.4. Random Forest (RF)

Unlike Bagging, Random Forest only considers a subset of features (mtry) at each split. For regression problems, the standard default is mtry = p/3 (where p is the number of markers). This decorrelates the trees and often improves performance on high-dimensional genomic data.

# 1. Setup
grid_rf <- expand.grid(Rep = 1:n_reps, Variable = target_vars, stringsAsFactors = FALSE)
out_rf <- "output/rf"
if (!dir.exists(out_rf)) dir.create(out_rf, recursive = TRUE)

cat("Iniciando RF. Total de tarefas:", nrow(grid_rf) * nfolds, "\n")

set.seed(123)

# 2. Execução
rf_list <- foreach(task_i = 1:nrow(grid_rf), 
                   .packages = c('randomForest', 'caret', 'tidyverse'),
                   .export = c("filesFenT", "filesGenT", "filesFenV", "filesGenV", "nfolds")) %dopar% {
  
  curr_rep <- grid_rf$Rep[task_i]
  curr_var <- grid_rf$Variable[task_i]
  
  # Acumuladores locais
  fold_metrics_accum <- data.frame()
  fold_importance_accum <- data.frame()
  
  for (fold_k in 1:nfolds) {
    
    # A. Lógica de Reciclagem de Arquivos
    raw_idx <- (curr_rep - 1) * nfolds + fold_k
    file_idx <- ((raw_idx - 1) %% length(filesFenT)) + 1
    
    # Checkpoint
    outfile_imp <- sprintf("%s/IMP_Rep%d_Var%d_Fold%d.csv", out_rf, curr_rep, curr_var, fold_k)
    if(file.exists(outfile_imp)) next
    
    tryCatch({
      # B. Leitura dos Dados
      dadosTy <- read.table(filesFenT[file_idx], header = FALSE)
      dadosTx <- read.table(filesGenT[file_idx], header = FALSE)
      dadosVy <- read.table(filesFenV[file_idx], header = FALSE)
      dadosVx <- read.table(filesGenV[file_idx], header = FALSE)
      
      start_time <- Sys.time()
      
      # C. Modelo RF Standard
      # mtry padrão para regressão é p/3
      rf_model <- randomForest(x = dadosTx, y = dadosTy[, curr_var], 
                               mtry = floor(ncol(dadosTx)/3), 
                               ntree = 500,
                               importance = TRUE)
      
      exec_time <- as.numeric(difftime(Sys.time(), start_time, units = "secs"))
      
      # --- D. CÁLCULO DE MÉTRICAS ---
      ypred <- predict(rf_model, newdata = dadosVx)
      ypred_train <- predict(rf_model)
      
      # Validação
      if (sd(ypred) == 0) { 
        r2_val <- 0 
      } else { 
        rv <- cor(dadosVy[, curr_var], ypred)
        r2_val <- as.numeric(rv^2)
      }
      rmse_val <- as.numeric(sqrt(mean((dadosVy[, curr_var] - ypred)^2)))
      
      # Treinamento
      rt <- cor(dadosTy[, curr_var], ypred_train)
      r2_train <- as.numeric(rt^2)
      rmse_train <- as.numeric(sqrt(mean((dadosTy[, curr_var] - ypred_train)^2)))
      
      # Salvar Métricas
      curr_metrics <- data.frame(
        Model = "RF", 
        Replicate = curr_rep, 
        Variable = curr_var, 
        Fold = fold_k, 
        R2_Train = r2_train, 
        R2_Val = r2_val, 
        RMSE_Train = rmse_train, 
        RMSE_Val = rmse_val, 
        Time_Sec = exec_time
      )
      
      write.csv(curr_metrics, sprintf("%s/METRICS_Rep%d_Var%d_Fold%d.csv", out_rf, curr_rep, curr_var, fold_k), row.names = FALSE)
      
      # --- E. IMPORTÂNCIA (Padrão Solicitado) ---
      
      imp <- tryCatch({ data.frame(varImp(rf_model)) }, error=function(e) NULL)
      
      if (!is.null(imp)) {
         # --- PADRONIZAÇÃO DE COLUNAS (IMPORTÂNCIA) ---
         imp$Marker <- rownames(imp)
         colnames(imp)[1] <- "Overall"
         
         imp_final <- data.frame(
            Model = "RF",
            Replicate = curr_rep,
            Variable = curr_var,
            Fold = fold_k,
            Marker = imp$Marker,
            Overall = imp$Overall
         )
         
         fold_importance_accum <- rbind(fold_importance_accum, imp_final)
         write.csv(imp_final, outfile_imp, row.names = FALSE)
      } else {
         dummy <- data.frame(Model="RF", Overall=0)
         write.csv(dummy, outfile_imp, row.names=FALSE)
      }
      
    }, error = function(e) {
       writeLines(paste("Erro no RF:", e$message), file.path(out_rf, "error_log.txt"))
    })
    
    # Limpeza de memória
    rm(rf_model, dadosTx, dadosVx, dadosTy, dadosVy); gc()
  }
  return(NULL)
}

4.5. Boosting (BO)

Gradient Boosting Machines (GBM) build an ensemble of weak prediction models (typically shallow trees) sequentially. Each new tree attempts to correct the errors (residuals) of the previous ones. Key parameters are shrinkage (learning rate) and interaction.depth (tree complexity).

# 1. Setup

  curr_var <- grid_bo$Variable[task_i]
  
  # Local accumulators (optional since we save to disk)
  fold_metrics_accum <- data.frame()
  fold_importance_accum <- data.frame()
  
  for (fold_k in 1:nfolds) {
    
    # A. Lógica de Reciclagem de Arquivos
    raw_idx <- (curr_rep - 1) * nfolds + fold_k
    file_idx <- ((raw_idx - 1) %% length(filesFenT)) + 1

      # B. Leitura dos Dados

5. Result Consolidation

This step recursively searches for all individual result files generated by the Machine Learning models. It consolidates them into unified datasets, ensuring strict data type consistency and cleaning Marker IDs (removing prefixes like “V” or “M”) to prevent merging errors.

cat("--- Starting Global ML Consolidation ---\n")

  # ==============================================================================

  imp_files <- list.files(path = ml_root_dir, pattern = "IMP_.*\\.csv", recursive = TRUE, full.names = TRUE)
  
  # Filter out any previously consolidated files
  imp_files <- imp_files[!grepl("CONSOLIDATED", imp_files)]
  
  if (length(imp_files) > 0) {

        df <- read.csv(f)

        if("Marker" %in% names(df)) {
          # 1. Forçar para texto (evita erro de integer vs character)

          df$Marker <- gsub("[^0-9]", "", df$Marker)
